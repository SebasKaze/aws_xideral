cols_pct = [col for col in datos_lista_top.columns if col.endswith('%')]

fig, ax = plt.subplots(figsize=(12, 6))

ancho = 0.12
x = range(len(datos_lista_top))

for i, col in enumerate(cols_pct):
    posiciones = [p + (i * ancho) for p in x]
    ax.bar(posiciones, datos_lista_top[col], width=ancho, label=col.replace('_%', '').capitalize())

ax.set_title("Características Musicales (Top 5 Canciones Más Escuchadas)", fontsize=14, pad=15)
ax.set_ylabel("Porcentaje (%)")
ax.set_ylim(0, 100)

ax.set_xticks([p + (ancho * (len(cols_pct) - 1) / 2) for p in x])
ax.set_xticklabels(datos_lista_top["track_name"], rotation=15, ha="right", fontsize=10)

ax.legend(title="Métricas", bbox_to_anchor=(1.05, 1), loc='upper left')
plt.tight_layout()

plt.show()
